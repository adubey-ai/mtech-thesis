\chapter{Mathematical Proofs}
\label{chp:appB}

This appendix collects the full proofs of the theoretical results
stated in the main text. The proofs are self-contained: each
reproduces the claim before giving the argument, and each is
numbered so that it can be cited from any chapter.

\section{Notation and Conventions}\label{app:notation}
Throughout this appendix we use the following conventions.
\begin{itemize}
    \item $N$ denotes the sequence length; $d$ the per-head embedding
          dimension; $H$ the number of heads; $B$ the batch size.
    \item $(B_r, B_c)$ are the row/column block sizes of Flash
          Attention; $T_r = \lceil N/B_r \rceil$ and
          $T_c = \lceil N/B_c \rceil$.
    \item $(w_l, w_r)$ are the left/right window half-widths; the full
          window is $w = w_l + w_r + 1$.
    \item $Q, K \in \mathbb{R}^{N \times d}$ and $V \in \mathbb{R}^{N
          \times d_v}$ are the query/key/value matrices.
    \item $A_{ij} \triangleq q_i \cdot k_j / \sqrt{d}$ is the raw
          attention score; the softmax-normalised weight is $P_{ij}$.
\end{itemize}

\section{Correctness of Corner-Based Block
Classification}\label{app:proofs_classify}

\begin{lemma}[Restatement of Lemma~\ref{lem:corner}]\label{lem:corner2}
Let $B = [r_0, r_1] \times [c_0, c_1]$ be a block of the attention
matrix. Define the window band
$W = \{(i,j) \in [0, N-1]^2 : -w_l \le j - i \le w_r\}$.
Then:
\begin{enumerate}
    \item[\textup{(i)}] $B \cap W = \emptyset$ iff
          $c_1 < r_0 - w_l$ \textbf{or} $c_0 > r_1 + w_r$;
    \item[\textup{(ii)}] $B \subseteq W$ iff
          $c_1 \le r_0 + w_r$ \textbf{and} $c_0 \ge r_1 - w_l$.
\end{enumerate}
\end{lemma}

\begin{proof}
Consider the scalar function $\phi(i, j) = j - i$ on $B$. Because
$\phi$ is linear and $B$ is a rectangle aligned with the coordinate
axes, $\phi$ attains its minimum at the corner $(r_1, c_0)$ (largest
$i$, smallest $j$) and its maximum at $(r_0, c_1)$ (smallest $i$,
largest $j$).

\textbf{(i) Outside case.} $B \cap W = \emptyset$ iff $\phi$ never
lies in $[-w_l, w_r]$ on $B$, which is equivalent to
$\max_B \phi < -w_l$ \emph{or} $\min_B \phi > w_r$. Substituting the
extrema,
$\max_B \phi = c_1 - r_0$ and $\min_B \phi = c_0 - r_1$, we get
$c_1 - r_0 < -w_l$ or $c_0 - r_1 > w_r$, i.e.\ $c_1 < r_0 - w_l$ or
$c_0 > r_1 + w_r$.

\textbf{(ii) Inside case.} $B \subseteq W$ iff $\phi(B) \subseteq
[-w_l, w_r]$, which requires $\max_B \phi \le w_r$ and
$\min_B \phi \ge -w_l$. Substituting,
$c_1 - r_0 \le w_r$ and $c_0 - r_1 \ge -w_l$, i.e.\
$c_1 \le r_0 + w_r$ and $c_0 \ge r_1 - w_l$.

The clipping of $W$ to $[0, N-1]^2$ does not change either inequality,
because $B$ itself is a subset of $[0, N-1]^2$ by construction.
\end{proof}

\begin{corollary}\label{cor:classifier_cost}
Algorithm~\ref{alg:swfa_classify} runs in $\mathcal{O}(T_r T_c)$
arithmetic operations and correctly assigns each block to exactly one
of the sets $\mathcal{I}, \mathcal{X}, \mathcal{O}$.
\end{corollary}

\begin{proof}
For each of the $T_r T_c$ blocks, Algorithm~\ref{alg:swfa_classify}
performs a constant number of comparisons derived from the two
predicates of Lemma~\ref{lem:corner2}. The three sets are disjoint by
construction: the \texttt{outside} predicate takes precedence; the
\texttt{inside} predicate is only tested when \texttt{outside} is
false; and the remaining blocks are placed in $\mathcal{X}$. By
Lemma~\ref{lem:corner2}, a block is outside iff \texttt{outside} is
true, and inside iff \texttt{inside} is true; the three sets are
therefore a partition of the grid.
\end{proof}

\section{Block-Count Lemma}\label{app:proofs_block}

\begin{lemma}[Skipped-block count]\label{lem:blockcount}
For a symmetric window $w_l = w_r = W$ and equal block sizes
$B_r = B_c = B$, with $B \mid N$, the number of outside blocks is
\begin{equation}\label{eq:block_count}
    |\mathcal{O}|
    \;=\; 2 \sum_{k=1}^{T_r - 1}
          \max(0, T_r - k - \lceil (W + (k-1) B) / B \rceil)
    \;=\; (T_r)^2 - \Theta(T_r W / B)
\end{equation}
and therefore the skip ratio converges to $1 - 2W / N$ as
$N \to \infty$ with fixed $W$.
\end{lemma}

\begin{proof}
With $B \mid N$, the grid has $T = N/B$ blocks per axis. By
Lemma~\ref{lem:corner2}(i), block $(r,c)$ is outside iff
$c B + B - 1 < r B - W$ or $c B > r B + B - 1 + W$, i.e.\
\begin{equation}\label{eq:blockout}
    c \le r - \left\lceil (W + B - 1)/B \right\rceil
    \quad \text{or} \quad
    c \ge r + \left\lceil (W + B - 1)/B \right\rceil.
\end{equation}
Let $k^\star = \lceil (W + B - 1)/B \rceil$. The number of outside
blocks in row $r$ is therefore $\max(0, r - k^\star) +
\max(0, T - 1 - r - k^\star)$, and summing over $r$ yields
$|\mathcal{O}| = 2 \sum_{r=0}^{T-1} \max(0, r - k^\star)$ after
collecting symmetric terms, which simplifies to
$T^2 - (2 k^\star - 1) T + k^\star(k^\star - 1)$.
Dividing by $T^2$ and letting $N \to \infty$ with $W$ fixed,
$k^\star / T \to W/N$, and the skip ratio converges to
$1 - 2W/N$, as claimed.
\end{proof}

\section{Convexity of the SWFA Cost}\label{app:proof_convex}

\begin{theorem}[Restatement of
Theorem~\ref{thm:swfa_convex}]\label{thm:convex2}
The cost function
\begin{equation}
    \mathcal{L}(w) = \alpha \kappa N w d + \beta M_0
        + \frac{2 \gamma N A_0}{1 - e^{-\lambda}}\, e^{-\lambda w / 2}
\end{equation}
is strictly convex for $w > 0$.
\end{theorem}

\begin{proof}
Differentiating twice,
\begin{equation}
    \mathcal{L}'(w) = \alpha \kappa N d
      - \frac{\gamma N A_0 \lambda}{1 - e^{-\lambda}}
        e^{-\lambda w / 2},
\end{equation}
\begin{equation}
    \mathcal{L}''(w)
      = \frac{\gamma N A_0 \lambda^2}{2(1 - e^{-\lambda})}
        e^{-\lambda w / 2}.
\end{equation}
All prefactors are positive: $\gamma > 0$ by assumption, $N, A_0,
\lambda > 0$ by model specification, and $1 - e^{-\lambda} > 0$
because $\lambda > 0$. The exponential $e^{-\lambda w / 2} > 0$ for
all $w \in \mathbb{R}$. Hence $\mathcal{L}''(w) > 0$ for all $w > 0$,
and $\mathcal{L}$ is strictly convex.
\end{proof}

\begin{remark}
Because $\mathcal{L}''$ is a positive exponential times a constant, the
convexity is \emph{strong}: for any $w_0 > 0$ there exists $\mu > 0$
such that $\mathcal{L}''(w) \ge \mu$ on any bounded interval
containing $w_0$. This gives a linear convergence rate for gradient
descent on $\mathcal{L}$ and justifies a Newton step for the
$w^\star$ derivation below.
\end{remark}

\section{Closed-Form Optimal Window}\label{app:proof_optw}

\begin{theorem}[Restatement of
Theorem~\ref{thm:swfa_optw}]\label{thm:optw2}
The minimiser of $\mathcal{L}$ is
\begin{equation}\label{eq:opt_w_app}
    w^\star = \frac{2}{\lambda}
      \ln\!\left(
      \frac{\gamma A_0 \lambda}{\alpha \kappa d (1 - e^{-\lambda})}
      \right),
\end{equation}
provided the argument of the logarithm exceeds one (otherwise the
unconstrained minimiser is non-positive and the projected minimiser
equals $1$).
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:convex2}, $\mathcal{L}$ is strictly convex on
$w > 0$, so its minimiser is the unique stationary point.
$\mathcal{L}'(w) = 0$ iff
\begin{equation}
    \alpha \kappa N d
    = \frac{\gamma N A_0 \lambda}{1 - e^{-\lambda}}
      e^{-\lambda w / 2},
\end{equation}
i.e.\
\begin{equation}
    e^{-\lambda w / 2}
    = \frac{\alpha \kappa d (1 - e^{-\lambda})}{\gamma A_0 \lambda}.
\end{equation}
Taking $\ln$ of both sides and multiplying by $-2/\lambda$
gives~\eqref{eq:opt_w_app}. The argument of the outer logarithm is
positive, so $w^\star \in \mathbb{R}$; it is positive exactly when the
argument exceeds one, which is the regime in which local attention
actually helps (otherwise dense attention is cheaper than the loss in
accuracy and $w = N - 1$ is optimal).

Projection onto $[1, N-1]$ preserves uniqueness: on the left, the
convexity implies $\mathcal{L}(1) \le \mathcal{L}(0)$, and on the right
$\mathcal{L}(N-1)$ is dominated by the computation term because the
exponential has decayed. Hence the projected minimiser is obtained by
clipping $w^\star$ to $[1, N-1]$.
\end{proof}

\begin{corollary}[Sensitivities]\label{cor:sens}
Let $w^\star$ be as in Theorem~\ref{thm:optw2}. Then:
\begin{enumerate}
    \item[\textup{(i)}] $\partial w^\star / \partial \alpha < 0$;
    \item[\textup{(ii)}] $\partial w^\star / \partial \gamma > 0$;
    \item[\textup{(iii)}] $\partial w^\star / \partial d < 0$;
    \item[\textup{(iv)}] $w^\star \to \infty$ as $\lambda \to 0^+$.
\end{enumerate}
\end{corollary}

\begin{proof}
Taking $\partial / \partial \alpha$
of~\eqref{eq:opt_w_app} gives
$\partial w^\star / \partial \alpha = -2/(\lambda \alpha) < 0$.
Similarly,
$\partial w^\star / \partial \gamma = 2/(\lambda \gamma) > 0$ and
$\partial w^\star / \partial d = -2/(\lambda d) < 0$. For (iv), expand
the logarithm for small $\lambda$: $1 - e^{-\lambda}
\approx \lambda - \lambda^2/2$, so the argument of the outer log tends
to $\gamma A_0 / (\alpha \kappa d)$, which is $O(1)$. The outer factor
$2/\lambda$ diverges, so $w^\star \to \infty$.
\end{proof}

\section{Flash Attention Softmax
Correctness}\label{app:proof_flash_softmax}

The online softmax update used by both Flash Attention and SWFA is
algebraically equivalent to the full-batch softmax. The proof is a
straightforward induction, reproduced for completeness.

\begin{lemma}[Online softmax equivalence]\label{lem:online_softmax}
Given scalars $s_1, \dots, s_N$ and values $v_1, \dots, v_N$, let
$m = \max_i s_i$,
$\ell = \sum_i e^{s_i - m}$, and $o = \sum_i e^{s_i - m} v_i / \ell$.
Then for any partition of the indices into $T$ blocks,
the iterative update
\begin{align}
    m^{(t)} &= \max(m^{(t-1)}, \max_{i \in B_t} s_i),\\
    \ell^{(t)} &= e^{m^{(t-1)} - m^{(t)}} \ell^{(t-1)}
                 + \sum_{i \in B_t} e^{s_i - m^{(t)}},\\
    o^{(t)} &= \frac{e^{m^{(t-1)} - m^{(t)}} \ell^{(t-1)} o^{(t-1)}
                      + \sum_{i \in B_t} e^{s_i - m^{(t)}} v_i}
                    {\ell^{(t)}},
\end{align}
with $(m^{(0)}, \ell^{(0)}, o^{(0)}) = (-\infty, 0, 0)$, yields
$(m^{(T)}, \ell^{(T)}, o^{(T)}) = (m, \ell, o)$.
\end{lemma}

\begin{proof}
By induction on $t$. Denote by $I_t = \bigcup_{s \le t} B_s$ the set
of indices processed up to step $t$, and define
$m_t = \max_{i \in I_t} s_i$,
$\ell_t = \sum_{i \in I_t} e^{s_i - m_t}$,
$o_t = \sum_{i \in I_t} e^{s_i - m_t} v_i / \ell_t$.

\emph{Base case.} $t = 0$: $I_0 = \emptyset$, all three quantities
coincide with the initialisation.

\emph{Inductive step.} Assume the claim holds for $t-1$. At step $t$,
$m^{(t)} = \max(m^{(t-1)}, \max_{i \in B_t} s_i) = m_t$ by definition.
Multiplying the previous sum by $e^{m^{(t-1)} - m^{(t)}}$ rescales the
exponents to the new reference $m^{(t)}$, and adding the contribution
from $B_t$ produces exactly
$\ell_t = \sum_{i \in I_t} e^{s_i - m^{(t)}}$, matching the claim. The
same rescaling applied to the numerator of $o$ gives $o_t$.
\end{proof}

\begin{corollary}[SWFA equivalence]
SWFA produces the same output as vanilla sliding-window attention up
to floating-point round-off.
\end{corollary}

\begin{proof}
SWFA skips exactly the blocks that are outside the window (by
Lemma~\ref{lem:corner2}(i)), and on the remaining blocks it applies
either no mask (inside) or the additive $\{0, -\infty\}$ mask of
Algorithm~\ref{alg:swfa_mask} (intersecting). Because masked entries
contribute zero to both the numerator and the denominator of the
softmax, the online update of Lemma~\ref{lem:online_softmax} acting
on masked scores is equivalent to the online update acting on the
full set of in-window scores. Hence SWFA's output equals the
vanilla sliding-window attention's output up to the rearrangement of
the summation order, and floating-point associativity accounts for
any observed differences.
\end{proof}

\section{Communication-Cost Bounds for
Distributed Strategies}\label{app:proof_comms}

\begin{lemma}[DDP ring-allreduce bandwidth]\label{lem:ddp_bw}
On a ring topology with $P$ workers and link bandwidth $B$, a single
all-reduce of a tensor of size $M s$ bytes takes
\begin{equation}\label{eq:ring_cost}
    T_{\text{ring}}(M, P) =
    2 (P - 1) \alpha + \frac{2 (P - 1) M s}{P B},
\end{equation}
where $\alpha$ is the per-hop latency and the factor $2(P-1)/P$
reflects the two phases (reduce-scatter + all-gather).
\end{lemma}

\begin{proof}
A ring all-reduce proceeds in two phases of $P - 1$ steps each. Each
step sends a fragment of size $M s / P$ to the next worker. The
latency term is therefore $2 (P - 1) \alpha$, and the bandwidth term
is $2 (P - 1) (M s / P) / B = 2 (P - 1) M s / (P B)$.
Summing gives~\eqref{eq:ring_cost}.
\end{proof}

\begin{corollary}\label{cor:ddp_limit}
As $P \to \infty$ with $M$ fixed, $T_{\text{ring}}(M, P) \to 2 M s / B
+ 2 (P - 1) \alpha$, so the per-worker bandwidth term is asymptotically
independent of $P$ and the scalability is limited only by the
latency constant.
\end{corollary}

\begin{lemma}[ZeRO-3 communication volume]\label{lem:zero3_bw}
ZeRO-3 incurs, per training step, a communication volume of
\begin{equation}
    V_{\text{ZeRO-3}} = \frac{2 (P - 1)}{P} \cdot M s
                     + \frac{2 (P - 1)}{P} \cdot M s
                     + \frac{2 (P - 1)}{P} \cdot M s_{\text{grad}}
\end{equation}
(two all-gathers for forward/backward parameters plus one reduce-scatter
for gradients).
\end{lemma}

\begin{proof}
Each parameter is sharded across $P$ workers. Before the forward
pass, each worker must receive the other workers' shards via an
all-gather, contributing $(P-1)/P \cdot M s$ per worker-input. By
symmetry the reverse all-gather before the backward pass adds a second
$(P-1)/P \cdot M s$; the reduce-scatter of the gradients is the
symmetric operation and contributes $(P-1)/P \cdot M s_{\text{grad}}$.
Summing gives the stated total.
\end{proof}

\begin{remark}
Comparing Lemma~\ref{lem:zero3_bw} with Lemma~\ref{lem:ddp_bw}, ZeRO-3
sends roughly $1.5\times$ the data of DDP per step. The overhead is
amortised by the $8\times$ reduction in memory, which allows a
correspondingly larger batch size and therefore fewer steps per
epoch.
\end{remark}

\section{Exponential Decay Assumption}\label{app:proof_decay}

The attention-decay assumption $A_{ij} = A_0 e^{-\lambda |i - j|}$ is
a modelling choice. We justify it from two directions.

\begin{proposition}[Empirical decay]\label{prop:decay}
For a transformer with softmax attention and inputs drawn from a
stationary Markov process with mixing time $\tau$, the expected
attention weight between tokens at distance $\Delta = |i - j|$
satisfies
$\mathbb{E}[P_{ij}] \le C \cdot e^{-\Delta / \tau}$ for some constant
$C > 0$.
\end{proposition}

\begin{proof}[Proof sketch]
By the data processing inequality for the Markov chain, the mutual
information $I(x_i; x_j)$ decays exponentially in $|i-j|$ with rate
$1/\tau$. Because softmax attention can be viewed as a non-parametric
estimator of $I(x_i; x_j)$ up to a constant, its expectation also
decays at rate $1/\tau$. Setting $\lambda = 1/\tau$ yields the
claim; a rigorous statement requires Dobrushin-style contraction
inequalities, which we omit.
\end{proof}

\begin{remark}
The closed-form optimal-window formula
\eqref{eq:opt_w_app} is robust to misspecification of $\lambda$: the
logarithm dampens parameter misestimation, so halving or doubling
$\lambda$ changes $w^\star$ by at most $\ln 2 \cdot 2/\lambda$, which
is typically $<\!20\%$ of the value itself.
\end{remark}

\section{Reproducibility of the Numerical
Accuracy Bound}\label{app:reproducibility}

\begin{proposition}[FP32 error bound for SWFA]\label{prop:fp32_err}
Let $u$ denote the FP32 unit round-off ($u \approx 6 \times 10^{-8}$).
The SWFA output $\hat O$ satisfies
\begin{equation}
    \|O - \hat O\|_\infty
    \le N (3 d + 2) u \cdot
        \|Q\|_\infty \|K\|_\infty \|V\|_\infty
        + \mathcal{O}(u^2).
\end{equation}
\end{proposition}

\begin{proof}[Proof sketch]
Each entry of $S_{rc} = Q_r K_c^\top / \sqrt{d}$ is the inner product
of two $d$-vectors and incurs relative error at most $d u$. The
online softmax adds at most two multiplications and one subtraction,
contributing $3 u$ per block pass; with $T_c$ column blocks and
sub-additive rounding, the total is $3 d + 2 T_c u \le 3 d + 2 N/B_c
u$. Multiplying by $V$ introduces another $d$-vector inner product,
bounded by $\|V\|_\infty$. Combining the bounds gives the claim up to
$\mathcal{O}(u^2)$ terms.
\end{proof}

Proposition~\ref{prop:fp32_err} predicts a worst-case FP32 error of
$\sim 4 \times 10^{-5}$ at $N = 128$, $d = 16$, which comfortably
bounds the measured $\sim 8 \times 10^{-7}$ in
Table~\ref{tab:swfa_accuracy}. The gap is explained by the typical
random-input pessimism of the analysis.
